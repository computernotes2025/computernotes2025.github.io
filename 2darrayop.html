Q1 — Row Sum and Column Sum

Input a square matrix and display the sum of each row and each column separately.
import java.util.*;
class RowColumnSum {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a[][] = new int[n][n];

        for(int i=0;i<n;i++)
            for(int j=0;j<n;j++)
                a[i][j] = sc.nextInt();

        for(int i=0;i<n;i++) {
            int s=0;
            for(int j=0;j<n;j++) s += a[i][j];
            System.out.println("Row " + (i+1) + " = " + s);
        }

        for(int j=0;j<n;j++) {
            int s=0;
            for(int i=0;i<n;i++) s += a[i][j];
            System.out.println("Column " + (j+1) + " = " + s);
        }
    }
}

Q2 — Largest and Smallest Element

Input a matrix and find the largest and smallest element along with their positions.
import java.util.*;
class MaxMin {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int r=sc.nextInt(), c=sc.nextInt();
        int a[][]=new int[r][c];

        for(int i=0;i<r;i++)
            for(int j=0;j<c;j++) a[i][j]=sc.nextInt();

        int max=a[0][0], min=a[0][0];
        int mr=0,mc=0,nr=0,nc=0;

        for(int i=0;i<r;i++)
            for(int j=0;j<c;j++) {
                if(a[i][j]>max){max=a[i][j];mr=i;mc=j;}
                if(a[i][j]<min){min=a[i][j];nr=i;nc=j;}
            }

        System.out.println("Largest = "+max+" at "+(mr+1)+","+(mc+1));
        System.out.println("Smallest = "+min+" at "+(nr+1)+","+(nc+1));
    }
}

Q3 — Diagonal Operations

Input a square matrix. Find the sum of the principal diagonal, secondary diagonal and their difference.
import java.util.*;
class DiagonalSum {
    public static void main(String args[]) {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt(), a[][]=new int[n][n];

        for(int i=0;i<n;i++)
            for(int j=0;j<n;j++) a[i][j]=sc.nextInt();

        int p=0,s=0;
        for(int i=0;i<n;i++) {
            p += a[i][i];
            s += a[i][n-1-i];
        }

        System.out.println("Principal diagonal = "+p);
        System.out.println("Secondary diagonal = "+s);
        System.out.println("Difference = "+Math.abs(p-s));
    }
}

Q7 — Matrix Multiplication

Input two compatible matrices and display their product.
import java.util.*;
class MatrixMultiplication {
    public static void main(String args[]) {
        Scanner sc=new Scanner(System.in);
        int r1=sc.nextInt(), c1=sc.nextInt();
        int r2=sc.nextInt(), c2=sc.nextInt();

        if(c1!=r2) {
            System.out.println("Multiplication not possible");
            return;
        }

        int a[][]=new int[r1][c1], b[][]=new int[r2][c2];

        for(int i=0;i<r1;i++)
            for(int j=0;j<c1;j++) a[i][j]=sc.nextInt();

        for(int i=0;i<r2;i++)
            for(int j=0;j<c2;j++) b[i][j]=sc.nextInt();

        for(int i=0;i<r1;i++) {
            for(int j=0;j<c2;j++) {
                int s=0;
                for(int k=0;k<c1;k++)
                    s += a[i][k]*b[k][j];
                System.out.print(s+" ");
            }
            System.out.println();
        }
    }
}

Q15 — Identity Matrix

Input a square matrix and check whether it is an identity matrix.
import java.util.*;
class Identity {
    public static void main(String args[]) {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt(), a[][]=new int[n][n];

        for(int i=0;i<n;i++)
            for(int j=0;j<n;j++) a[i][j]=sc.nextInt();

        boolean flag=true;
        for(int i=0;i<n;i++)
            for(int j=0;j<n;j++) {

        else System.out.println("Not an Identity Matrix");
    }
}

Q16 — Sparse Matrix

Input a matrix and determine whether it is sparse. A matrix is sparse if the number of zeroes is greater than the number of non-zero elements.
import java.util.*;
class Sparse {
    public static void main(String args[]) {
        Scanner sc=new Scanner(System.in);
        int r=sc.nextInt(), c=sc.nextInt();
        int a[][]=new int[r][c];
        int z=0,nz=0;

        for(int i=0;i<r;i++)
            for(int j=0;j<c;j++) {
                a[i][j]=sc.nextInt();
                if(a[i][j]==0) z++;
                else nz++;
            }

        if(z>nz) System.out.println("Sparse Matrix");
        else System.out.println("Not a Sparse Matrix");
    }
}

Q25 — Saddle Point

Input a matrix and find the saddle point, if present. A saddle point is an element that is the smallest in its row but largest in its column.
import java.util.*;
class SaddlePoint {
    public static void main(String args[]) {
        Scanner sc=new Scanner(System.in);
        int r=sc.nextInt(), c=sc.nextInt();
        int a[][]=new int[r][c];
        boolean found=false;

        for(int i=0;i<r;i++)
            for(int j=0;j<c;j++) a[i][j]=sc.nextInt();

        for(int i=0;i<r;i++) {
            int min=a[i][0];

            for(int j=1;j<c;j++)
                if(a[i][j]<min) min=a[i][j];

            for(int j=0;j<c;j++) {
                if(a[i][j]==min) {
                    boolean largest=true;

                    for(int k=0;k<r;k++)
                        if(a[k][j]>min) largest=false;

                    if(largest) {
                        System.out.println("Saddle point = "+min);
                        found=true;
                    }
                }
            }
        }

        if(!found) System.out.println("No saddle point");
    }
}
