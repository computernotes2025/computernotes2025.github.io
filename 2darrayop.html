Q1 — Row Sum and Column Sum

Input a square matrix and display the sum of each row and each column separately. 
class RowColumnSum {
    static void calculate(int[][] a) {
        int n = a.length;
        
        for(int i=0; i<n; i++) {
            int s = 0;
            for(int j=0; j<n; j++) s += a[i][j];
            System.out.println("Row " + (i+1) + " = " + s);
        }

        for(int j=0; j<n; j++) {
            int s = 0;
            for(int i=0; i<n; i++) s += a[i][j];
            System.out.println("Column " + (j+1) + " = " + s);
        }
    }
}

Q2 — Largest and Smallest Element

Input a matrix and find the largest and smallest element along with their positions. 
class MaxMin {
    static void find(int[][] a) {
        int r = a.length, c = a[0].length;
        int max = a[0][0], min = a[0][0];
        int mr = 0, mc = 0, nr = 0, nc = 0;

        for(int i=0; i<r; i++) {
            for(int j=0; j<c; j++) {
                if(a[i][j] > max) { max = a[i][j]; mr = i; mc = j; }
                if(a[i][j] < min) { min = a[i][j]; nr = i; nc = j; }
            }
        }

        System.out.println("Largest = " + max + " at " + (mr+1) + "," + (mc+1));
        System.out.println("Smallest = " + min + " at " + (nr+1) + "," + (nc+1));
    }
}

Q3 — Diagonal Operations

Input a square matrix. Find the sum of the principal diagonal, secondary diagonal and their difference. 
class DiagonalSum {
    static void calculate(int[][] a) {
        int n = a.length;
        int p = 0, s = 0;
        
        for(int i=0; i<n; i++) {
            p += a[i][i];
            s += a[i][n-1-i];
        }

        System.out.println("Principal diagonal = " + p);
        System.out.println("Secondary diagonal = " + s);
        System.out.println("Difference = " + Math.abs(p-s));
    }
}

Q7 — Matrix Multiplication

Input two compatible matrices and display their product. 
class MatrixMultiplication {
    static void multiply(int[][] a, int[][] b) {
        int r1 = a.length, c1 = a[0].length;
        int r2 = b.length, c2 = b[0].length;

        if(c1 != r2) {
            System.out.println("Multiplication not possible");
            return;
        }

        for(int i=0; i<r1; i++) {
            for(int j=0; j<c2; j++) {
                int s = 0;
                for(int k=0; k<c1; k++) {
                    s += a[i][k] * b[k][j];
                }
                System.out.print(s + " ");
            }
            System.out.println();
        }
    }
}

Q8 — Interchange Rows

Input a matrix and interchange the first row with the last row. 
class InterchangeRows {
    static void swap(int[][] a) {
        int r = a.length, c = a[0].length;
        
        for(int j=0; j<c; j++) {
            int t = a[0][j];
            a[0][j] = a[r-1][j];
            a[r-1][j] = t;
        }

        for(int i=0; i<r; i++) {
            for(int j=0; j<c; j++) System.out.print(a[i][j] + " ");
            System.out.println();
        }
    }
}

class InterchangeColumns {

        for(int i=0; i<r; i++) {
            int t = a[i][0];
