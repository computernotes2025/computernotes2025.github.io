Factorial Implementation Methods

Recursive

Method 1: Classic Direct

int factorial(int n) {
    if (n == 0) return 1;
    else return n * factorial(n - 1);
}
Shortest and standard ISC answer.
Recursive

Method 2: State Accumulator

int f = 1;
int factorial(int n) {
    if (n == 0) return f;
    f = f * n;
    return factorial(n - 1);
}
Multiplies during call-stack push.
Recursive

Method 3: Upward Counter

int f = 1;
int factorial(int n, int i) {
    if (i <= n) {
        f = f * i;
        return factorial(n, i + 1);
    }
    return f;
}
Iterative style using counter i.
Iterative

Iterative Fallback

int factorial(int n) {
    int f = 1;
    for (int i = 1; i <= n; i++) f *= i;
    return f;
}
Non-recursive comparison loop.

Permutations & Combinations

Permutation P(n, r)

int perm(int n, int r) {
    return factorial(n) / factorial(n - r);
}
nPr = n! / (n-r)!

Combination C(n, r)

int comb(int n, int r) {
    return factorial(n) / (factorial(r) * factorial(n - r));
}
nCr = n! / (r! × (n-r)!)

Full ISC Class

class Permutation {
    int n, r;
    int factorial(int num) {
        if (num == 0) return 1;
        return num * factorial(num - 1);
    }
    void compute() { 
        int res = factorial(n) / factorial(n - r);
        System.out.println("P("+n+","+r+") = "+res);
    }
}

Special Number

void isSpecial(int n) {

        d = n % 10;

Power & Series

Recursive

Power a^b

double power(int a, int b) {
    if (b == 0) return 1.0;
    return a * power(a, b - 1);
}
Iterative

Power Iterative

double power(int a, int b) {
    double p = 1;
    for (int i=1; i<=b; i++) p *= a;
    return p;
}

Series Calculation

void calcSeries(int x, int n) {
    double sum = 0;
    for(int i=0; i<n; i++) {
        sum += power(x,i) / factorial(i);
    }
}

Fibonacci

nth Term Direct

int fib(int n) {
    if (n <= 1) return 0;
    else if (n == 2) return 1;
    else return fib(n-1)+fib(n-2);
}
Dual-recursion tree structure.

Generate First N

void printFib(int limit) {
    for(int i=1; i<=limit; i++)
        System.out.print(fib(i) + " ");
}

Tail Recursion

int a=0,b=1,c=0;
void fib(int i, int limit) {
    if(i <= limit) {
        c = a + b;
        System.out.print(c + " ");
        a = b; b = c; 
        fib(i + 1, limit);
    }
}
Efficient linear generator.

GCD & LCM

GCD Modulo

int gcd(int p, int q) {
    if (q == 0) return p;
    return gcd(q, p % q);
}
Euclidean algorithm.

GCD Subtraction

int gcd(int p, int q) {
    if (p == q) return p;
    if (p > q) return gcd(p - q, q);
    return gcd(p, q - p);
}

LCM Incremental

int lcm = 1;
int calcLCM(int a, int b) {
    if (lcm % a == 0 && lcm % b == 0) return lcm;
    lcm++;
    return calcLCM(a, b);
}

Prime & Emirp

Factor Counting

int count = 0;
int prime(int n, int i) {
    if (i <= n) { 
        if (n % i == 0) count++; 
        return prime(n, i + 1); 
    }
    return count;
}

Early Exit

int isPrime(int n, int i) {
    if (i == n) return 1;
    if (n % i == 0 || n == 1) return 0;

Digits & Magic

Sum of Digits

int sumDig(int n) {
    if(n == 0) return 0;
    return (n % 10) + sumDig(n / 10);
}

LIFO Print

Magic Number

void isMagic(int n) {
    while(n > 9) n = sumDig(n);

Happy Number

int sumSqDigits(int x) {
    if(x == 0) return 0;
    int d = x % 10; 
    return d*d + sumSqDigits(x/10);
}

Arrays & Sorting

Linear Search

int linearSearch(int A[], int i, int key) {
    if(i == A.length) return -1;
    if(A[i] == key) return 1;
    return linearSearch(A, i + 1, key);
}

Binary Search

int binSearch(int A[], int l, int u, int key) {
    if(u < l) return -1;
    int mid = (l + u) / 2;
    if(A[mid] == key) return 1;
    if(key > A[mid]) return binSearch(A, mid + 1, u, key);
    return binSearch(A, l, mid - 1, key);
}

Selection Sort

void selectionSort(int A[], int i) {
    if(i < A.length - 1) {
        for(int j = i + 1; j < A.length; j++) 
            if(A[i] > A[j]) swap(A, i, j);
        selectionSort(A, i + 1);
    }
}

Bubble Sort

void bubbleSort(int A[], int i) {
    if(i < A.length - 1) {
        for(int j = 0; j < A.length - 1 - i; j++) 
            if(A[j] > A[j+1]) swap(A, j, j+1);
        bubbleSort(A, i + 1);
    }
}
